{{- /* A click-to-load video (the youtube and vimeo shortcodes). Argument: dict "ctx" (the shortcode) "service". */ -}} {{- $c := .ctx -}} {{- /* The id is used as text: a bare number ({{< vimeo 253289453 >}}) arrives as an integer, so it is turned into a string first, and anything that is not a plain video id stops the build. */ -}} {{- $id := string (cond $c.IsNamedParams ($c.Get "id") ($c.Get 0) | default "") -}} {{- if not $id }}{{ errorf "%s shortcode without a video id in %s" .service $c.Page.File.Path }}{{ end -}} {{- if not (findRE `^[A-Za-z0-9_-]+$` $id) }}{{ errorf "%s shortcode: %q is not a video id (in %s)" .service $id $c.Page.File.Path }}{{ end -}} {{- $n := add ($c.Page.Store.Get "videos" | default 0) 1 }}{{ $c.Page.Store.Set "videos" $n -}} {{- $title := cond $c.IsNamedParams ($c.Get "title") "" -}} {{- $label := $title | default (printf "Video %d on the page “%s”" $n $c.Page.Title) -}} {{- $start := cond $c.IsNamedParams ($c.Get "start") "" -}} {{- $embed := "" }}{{ $watch := "" }}{{ $name := "" -}} {{- if eq .service "youtube" -}} {{- $embed = printf "https://www.youtube-nocookie.com/embed/%s?autoplay=1&rel=0%s" $id (cond (ne $start "") (printf "&start=%v" $start) "") -}} {{- $watch = printf "https://www.youtube.com/watch?v=%s%s" $id (cond (ne $start "") (printf "&t=%vs" $start) "") }}{{ $name = "YouTube" -}} {{- else -}} {{- $embed = printf "https://player.vimeo.com/video/%s?dnt=1&autoplay=1" $id -}} {{- $watch = printf "https://vimeo.com/%s" $id }}{{ $name = "Vimeo" -}} {{- end -}} {{- /* the poster: a still of the video saved next to the page as video-.jpg, shown as resized WebP; nothing is loaded from the video site before the visitor presses play */ -}} {{- $poster := "" -}} {{- with $c.Page.Resources.Get (printf "video-%s.jpg" $id) -}} {{- $set := slice -}} {{- $p := . }}{{ range slice 480 800 1280 }}{{ if le . $p.Width }}{{ $r := $p.Resize (printf "%dx webp q78" .) }}{{ $set = $set | append (printf "%s %dw" $r.RelPermalink $r.Width) }}{{ end }}{{ end -}} {{- $fb := $p.Resize "800x" -}} {{- $poster = printf `` $fb.RelPermalink (delimit $set ", ") $fb.Width $fb.Height -}} {{- end -}}
{{ $poster | safeHTML }} {{ $title | default "Play the video" }}The player loads from {{ $name }} when you start it.
