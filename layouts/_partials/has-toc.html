{{- /* Whether a page shows a table of contents: front matter toc: true / false decides; otherwise a page with at least site.Params.tocMinHeadings headings of level 1 to 3 gets one. Returns a bool. */ -}} {{- $n := 0 -}} {{- range .Fragments.HeadingsMap }}{{ if and (le .Level 3) (strings.TrimSpace .Title) }}{{ $n = add $n 1 }}{{ end }}{{ end -}} {{- $on := ge $n (site.Params.tocMinHeadings | default 2) -}} {{- if isset .Params "toc" }}{{ $on = .Params.toc }}{{ end -}} {{- return (and $on (gt $n 1)) -}}
