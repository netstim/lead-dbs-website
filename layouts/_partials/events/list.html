{{- /* All events of data/events.yaml, sorted by start, each with what the templates print: .e (the entry), .id, .start / .ends (time.Time; ends is exclusive), .iso_start / .iso_ends, .when (date text), .time (time with its zone, or ""), .time_short (the same, zone abbreviation only), .over (true when it ended before this build). Returns a slice of dicts. */ -}} {{- $out := slice -}} {{- range hugo.Data.events.events -}} {{- $tz := .timezone | default "Europe/Berlin" -}} {{- $s := string .start -}}{{- if eq (len $s) 16 }}{{ $s = printf "%s:00" $s }}{{ end -}} {{- $timed := gt (len $s) 10 -}} {{- $start := time.AsTime $s $tz -}} {{- $day := time.AsTime (substr $s 0 10) $tz -}} {{- $ends := $day.AddDate 0 0 1 -}} {{- $lastDay := $day -}} {{- with .end -}} {{- $e := string . -}}{{- if eq (len $e) 16 }}{{ $e = printf "%s:00" $e }}{{ end -}} {{- if gt (len $e) 10 -}}{{- $ends = time.AsTime $e $tz -}}{{- $lastDay = time.AsTime (substr $e 0 10) $tz -}} {{- else -}}{{- $lastDay = time.AsTime $e $tz -}}{{- $ends = $lastDay.AddDate 0 0 1 -}}{{- end -}} {{- end -}} {{- $when := "" -}} {{- if eq ($day.Format "2006-01-02") ($lastDay.Format "2006-01-02") -}} {{- $when = $day.Format (cond $timed "Monday 2 January 2006" "2 January 2006") -}} {{- else if eq ($day.Format "2006-01") ($lastDay.Format "2006-01") -}} {{- $when = printf "%s–%s" ($day.Format "2") ($lastDay.Format "2 January 2006") -}} {{- else if eq $day.Year $lastDay.Year -}} {{- $when = printf "%s – %s" ($day.Format "2 January") ($lastDay.Format "2 January 2006") -}} {{- else -}} {{- $when = printf "%s – %s" ($day.Format "2 January 2006") ($lastDay.Format "2 January 2006") -}} {{- end -}} {{- /* the zone is printed from the date and the IANA zone, so a summer date reads CEST by itself: "15:00 CET (Berlin time)" in full, "15:00 CET" where space is short. tz_label overrides both. */ -}} {{- $time := "" }}{{ $timeShort := "" -}} {{- if $timed -}} {{- $parts := split $tz "/" -}} {{- $city := replace (index $parts (sub (len $parts) 1)) "_" " " -}} {{- $abbr := $start.Format "MST" -}} {{- $time = printf "%s %s" ($start.Format "15:04") (.tz_label | default (printf "%s (%s time)" $abbr $city)) -}} {{- $timeShort = printf "%s %s" ($start.Format "15:04") (.tz_label | default $abbr) -}} {{- end -}} {{- $iso := "2006-01-02T15:04:05Z07:00" -}} {{- $out = $out | append (dict "e" . "id" (printf "%s-%s" (urlize .title) ($day.Format "2006-01-02")) "start" $start "ends" $ends "start_unix" $start.Unix "iso_start" ($start.Format $iso) "iso_ends" ($ends.Format $iso) "date_only" ($day.Format "2006-01-02") "timed" $timed "tz" $tz "when" $when "time" $time "time_short" $timeShort "over" (not (now.Before $ends))) -}} {{- end -}} {{- return (sort $out "start_unix") -}}
