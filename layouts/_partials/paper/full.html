{{- /* One paper of data/papers.yaml as a full reference: authors (year). Title. Journal, volume(issue), pages. doi link. Context: the paper's map. */ -}} {{- $p := . -}} {{- $title := $p.title -}} {{- if not (findRE `[.?!]$` $title) }}{{ $title = printf "%s." $title }}{{ end -}} {{- $out := printf "%s (%v). %s %s" $p.authors $p.year $title $p.journal -}} {{- with $p.volume }}{{ $out = printf "%s, %s" $out . }}{{ end -}} {{- with $p.issue }}{{ $out = printf "%s(%s)" $out . }}{{ end -}} {{- with $p.pages }}{{ $out = printf "%s, %s" $out . }}{{ end -}} {{- with $p.note }}{{ $out = printf "%s (%s)" $out . }}{{ end -}} {{- $out = printf "%s. doi:%s" $out $p.doi $p.doi -}} {{- $out | safeHTML -}}
