{{- /* data/papers.yaml, checked: the shortcodes papers and paper and the partial cite-box read it through this partial (partialCached, so the checks run once per build). Stops the build with the paper's id when a paper has an unknown field or type, lacks one of short, group, authors or title, lacks the journal of a listed group's article, has a DOI that is not of the form 10.xxxx/…, has a url without a site to name it, or names a group that is not under "groups": Publications prints only those groups, so the paper would vanish from it without a word. Returns the data (groups, papers). */ -}} {{- $d := hugo.Data.papers -}} {{- $groups := slice }}{{ $listed := slice }}{{ range $d.groups }}{{ $groups = $groups | append .id }}{{ if not (and (isset . "listed") (not .listed)) }}{{ $listed = $listed | append .id }}{{ end }}{{ end -}} {{- $fields := slice "short" "group" "type" "authors" "year" "title" "journal" "edition" "publisher" "address" "volume" "issue" "pages" "note" "doi" "url" "site" "links" -}} {{- range $id, $p := $d.papers -}} {{- range $k, $_ := $p }}{{ if not (in $fields $k) }}{{ errorf "data/papers.yaml: %s has an unknown field %q" $id $k }}{{ end }}{{ end -}} {{- range slice "short" "group" "authors" "title" }}{{ if not (index $p .) }}{{ errorf "data/papers.yaml: %s has no %s" $id . }}{{ end }}{{ end -}} {{- if not (in (slice "article" "inproceedings" "book" "misc") ($p.type | default "article")) }}{{ errorf "data/papers.yaml: %s: type %q is not one of article, inproceedings, book, misc" $id $p.type }}{{ end -}} {{- if and (in $listed $p.group) (not (and $p.journal $p.doi)) }}{{ errorf "data/papers.yaml: %s is listed on Publications, so it needs a journal and a doi" $id }}{{ end -}} {{- if and $p.url (not $p.site) }}{{ errorf "data/papers.yaml: %s: a url needs a site, the link's text (e.g. PubMed)" $id }}{{ end -}} {{- range $p.links }}{{ if not (and .label .url) }}{{ errorf "data/papers.yaml: %s: each of links needs a label and a url" $id }}{{ end }}{{ end -}} {{- with $p.doi }}{{ if not (findRE `^10\.\d{4,9}/\S+$` (string .)) }}{{ errorf "data/papers.yaml: %s: doi %q is not of the form 10.1016/j.neuroimage.2023.119862 (no https://doi.org/ before it)" $id . }}{{ end }}{{ end -}} {{- with $p.group }}{{ if not (in $groups .) }}{{ errorf "data/papers.yaml: %s: group %q is not one of the groups (%s), so Publications would leave it out" $id . (delimit $groups ", ") }}{{ end }}{{ end -}} {{- end -}} {{- return $d -}}
